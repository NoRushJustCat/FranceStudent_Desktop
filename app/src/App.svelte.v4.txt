<script>
  import { onMount, tick } from 'svelte';
  import './extra.css';
  import './ui.css';
  import * as api from './lib/api.js';
  import Message from './lib/Message.svelte';
  import Tool from './lib/Tool.svelte';
  import Settings from './lib/Settings.svelte';

  const CODE_AGENT = -2003;

  // ---- état ----
  let settings = $state(null);
  let agents = $state([]);
  let agentId = $state(null);
  let pickedAgent = false;
  let chatAgent = null;
  let chats = $state([]);
  let chatId = $state(null);
  let items = $state([]);
  let input = $state('');
  let busy = $state(false);
  let error = $state('');
  let mode = $state('chat'); // 'chat' | 'code'
  let workdir = $state(null);
  let autoAccept = $state(false);
  let eco = $state(false);
  let usage = $state(null);
  let ctx = $state(null);
  let skills = $state([]);
  let mcpUp = $state(0);
  let last = $state(null); // dernier message : tokens et appels
  let session = $state({ calls: 0, prompt: 0, output: 0, messages: 0 });

  let showSettings = $state(false);
  let settingsTab = $state('compte');
  let renaming = $state(null);
  let renameText = $state('');
  let confirmDel = $state(null);

  let box;
  let ta;

  // ---- utilitaires tolérants (formats API pas tous connus) ----
  const asArray = (r, keys) => (Array.isArray(r) ? r : (keys.map((k) => r?.[k]).find(Array.isArray) ?? []));
  const num = (v) => (typeof v === 'number' && Number.isFinite(v) ? v : null);
  const fmt = (n) => (n == null ? '–' : n >= 1000 ? (n / 1000).toFixed(n >= 10000 ? 0 : 1) + 'k' : String(n));
  const baseName = (p) => (p ? (p.split(/[\\/]/).filter(Boolean).pop() ?? p) : '');

  function readBudget(u) {
    const b = u?.chatBudget ?? u?.budget ?? u;
    if (!b || typeof b !== 'object') return null;
    let pct = null;
    for (const k of ['usedPercent', 'percentUsed', 'usagePercent', 'percent']) {
      if (num(b[k]) != null) {
        pct = b[k];
        break;
      }
    }
    if (pct == null) {
      const used = num(b.usedUsd ?? b.used);
      const lim = num(b.limitUsd ?? b.limit);
      const rem = num(b.remainingUsd ?? b.remaining);
      if (used != null && lim) pct = (used / lim) * 100;
      else if (rem != null && lim) pct = (1 - rem / lim) * 100;
    }
    if (pct == null) return null;
    return {
      pct: Math.max(0, Math.min(100, pct)),
      uncertain: !!(b.accountingUncertain ?? u?.accountingUncertain),
    };
  }
  function readCtx(c) {
    if (!c || typeof c !== 'object') return null;
    const pick = (ks) => ks.map((k) => num(c[k])).find((v) => v != null) ?? null;
    const used = pick(['promptTokens', 'tokens', 'usedTokens', 'contextTokens', 'totalTokens', 'used']);
    if (used == null) return null;
    return { used, max: pick(['maxTokens', 'contextWindow', 'contextSize', 'limit', 'max']) };
  }
  function toItems(chat) {
    const arr = chat?.chatMessages ?? chat?.messages ?? [];
    return arr
      .map((m) => {
        const role = String(m.role ?? m.author ?? m.sender ?? '').toLowerCase();
        let text = m.content ?? m.text ?? '';
        if (Array.isArray(text)) text = text.map((p) => p?.text ?? '').join('');
        return {
          kind: role.startsWith('user') || role === 'human' ? 'user' : 'assistant',
          text: String(text ?? ''),
          reasoning: '',
        };
      })
      .filter((i) => i.text);
  }

  const budget = $derived(readBudget(usage));
  const ctxInfo = $derived(ctx);
  const modeLabel = $derived(mode === 'code' ? 'Code' : 'Chat');

  // ---- suggestions de skills ($nom) ----
  const skillQuery = $derived.by(() => {
    const m = /(?:^|\s)\$([\w.-]*)$/.exec(input);
    return m ? m[1].toLowerCase() : null;
  });
  const suggestions = $derived(
    skillQuery === null ? [] : skills.filter((s) => s.name.toLowerCase().includes(skillQuery)).slice(0, 6),
  );
  function pickSkill(name) {
    input = input.replace(/\$([\w.-]*)$/, () => '$' + name + ' ');
    ta?.focus();
  }
  function insertSkill(name) {
    const sep = input && !/\s$/.test(input) ? ' ' : '';
    input = input + sep + '$' + name + ' ';
    ta?.focus();
  }

  // ---- chargement ----
  async function loadChats() {
    try {
      const r = await api.listChats(40, null);
      chats = asArray(r, ['chats', 'items', 'data']).map((c) => ({
        id: String(c.id ?? c.chatId),
        title: c.title ?? c.name ?? 'Sans titre',
      }));
    } catch (e) {
      error = String(e);
    }
  }
  async function loadUsage() {
    try {
      usage = await api.getUsage();
    } catch {
      /* jauge facultative */
    }
  }
  async function loadCtx() {
    if (!chatId) {
      ctx = null;
      return;
    }
    try {
      ctx = readCtx(await api.getContext(chatId));
    } catch {
      /* facultatif */
    }
  }
  async function loadMcp() {
    try {
      const s = await api.mcpStatus();
      mcpUp = s.servers.filter((x) => x.connected).length;
    } catch {
      /* facultatif */
    }
  }
  async function loadSkills() {
    try {
      skills = await api.listSkills(mode === 'code' ? workdir : null);
    } catch {
      skills = [];
    }
  }
  async function loadAll() {
    try {
      agents = (await api.listAgents()).filter((a) => a.isActive !== false);
      const saved = settings?.lastAgentId;
      pickedAgent = saved != null && agents.some((a) => a.id === saved);
      agentId = pickedAgent ? saved : (agents.find((a) => a.isDefault)?.id ?? agents[0]?.id ?? null);
    } catch (e) {
      error = String(e);
    }
    await Promise.all([loadChats(), loadUsage(), loadSkills(), loadMcp()]);
  }
  async function loadSettings() {
    settings = await api.getSettings();
    eco = settings.eco;
    autoAccept = settings.autoAccept;
    workdir = settings.lastWorkdir ?? null;
  }

  onMount(async () => {
    try {
      await loadSettings();
    } catch (e) {
      error = String(e);
      return;
    }
    if (!settings.hasKey) {
      openSettings('compte');
      return;
    }
    await loadAll();
  });

  $effect(() => {
    // recharge les skills quand le dossier ou le mode change
    mode;
    workdir;
    loadSkills();
  });

  async function onSaved() {
    showSettings = false;
    await loadSettings();
    await loadAll();
  }

  // ---- chats ----
  function newChat() {
    if (busy) return;
    chatId = null;
    items = [];
    ctx = null;
    error = '';
  }
  async function openChat(id) {
    if (busy || id === chatId) return;
    error = '';
    chatId = id;
    items = [];
    try {
      const chat = await api.getChat(id);
      items = toItems(chat);
      const aid = chat?.agent?.id;
      if (aid != null && agents.some((a) => a.id === aid)) agentId = aid;
      await tick();
      if (box) box.scrollTop = box.scrollHeight;
    } catch (e) {
      error = String(e);
    }
    loadCtx();
  }
  function startRename(c) {
    renaming = c.id;
    renameText = c.title;
  }
  async function renameKey(e, c) {
    if (e.key === 'Escape') renaming = null;
    if (e.key !== 'Enter') return;
    const title = renameText.trim();
    renaming = null;
    if (!title || title === c.title) return;
    try {
      await api.renameChat(c.id, title);
      chats = chats.map((x) => (x.id === c.id ? { ...x, title } : x));
    } catch (err) {
      error = String(err);
    }
  }
  async function del(c) {
    if (confirmDel !== c.id) {
      confirmDel = c.id;
      return;
    }
    confirmDel = null;
    try {
      await api.deleteChat(c.id);
      chats = chats.filter((x) => x.id !== c.id);
      if (chatId === c.id) newChat();
    } catch (e) {
      error = String(e);
    }
  }
  const focus = (n) => {
    n.focus();
    n.select();
  };

  // ---- barre du haut ----
  function setMode(m) {
    if (busy || m === mode) return;
    if (m === 'code') {
      chatAgent = agentId;
      if (!pickedAgent && agents.some((a) => a.id === CODE_AGENT)) agentId = CODE_AGENT;
    } else if (chatAgent != null) {
      agentId = chatAgent;
    }
    mode = m;
  }
  function onAgent() {
    pickedAgent = true;
    api.saveSettings({ lastAgentId: agentId });
  }
  async function chooseFolder() {
    try {
      const p = await api.pickFolder();
      if (p) {
        workdir = p;
        await api.saveSettings({ lastWorkdir: p });
      }
    } catch (e) {
      error = String(e);
    }
  }
  function setAuto(v) {
    autoAccept = !!v;
    api.saveSettings({ autoAccept }).catch((e) => (error = String(e)));
  }
  function onAuto(e) {
    setAuto(e.currentTarget.checked);
  }
  function setEco(v) {
    eco = !!v;
    api.saveSettings({ eco }).catch((e) => (error = String(e)));
  }
  function openSettings(tab = 'compte') {
    settingsTab = tab;
    showSettings = true;
  }
  function onInsertSkill(name) {
    insertSkill(name);
    showSettings = false;
  }
  function closeSettings() {
    showSettings = false;
    loadSkills();
    loadMcp();
  }
  async function compact() {
    if (!chatId || busy) return;
    try {
      await api.compactChat(chatId);
      await loadCtx();
    } catch (e) {
      error = String(e);
    }
  }
  function resetSession() {
    session = { calls: 0, prompt: 0, output: 0, messages: 0 };
    last = null;
  }

  // ---- envoi ----
  function stick() {
    if (!box) return;
    const near = box.scrollHeight - box.scrollTop - box.clientHeight < 120;
    if (near) tick().then(() => (box.scrollTop = box.scrollHeight));
  }

  async function resolve(t, allow, always) {
    try {
      if (allow && always && t.origin === 'mcp') await api.trustTool(t.name, true);
      t.status = allow ? 'running' : 'denied';
      await api.resolveTool(t.callId, allow);
    } catch (e) {
      error = String(e);
    }
  }

  async function send() {
    const content = input.trim();
    if (!content || busy) return;
    if (mode === 'code' && !workdir) {
      error = 'Choisis d’abord un dossier de travail (bouton « Dossier »).';
      return;
    }
    if (agentId == null) {
      error = 'Aucun agent disponible.';
      return;
    }
    input = '';
    error = '';
    busy = true;
    items.push({ kind: 'user', text: content });
    if (box) tick().then(() => (box.scrollTop = box.scrollHeight));

    let cur = null;
    const assistant = () => {
      if (!cur) {
        items.push({ kind: 'assistant', text: '', reasoning: '' });
        cur = items[items.length - 1];
      }
      return cur;
    };

    try {
      await api.sendMessage(
        {
          chatId,
          agentId,
          content,
          workdir: mode === 'code' ? workdir : null,
          autoAccept,
          eco,
        },
        (ev) => {
          switch (ev.type) {
            case 'opened':
              chatId = ev.chat_id;
              if (!chats.some((c) => c.id === ev.chat_id)) {
                chats = [{ id: ev.chat_id, title: 'Nouveau chat' }, ...chats];
              }
              break;
            case 'title':
              chats = chats.map((c) => (c.id === ev.chat_id ? { ...c, title: ev.title } : c));
              break;
            case 'delta':
              stick();
              assistant().text += ev.text;
              break;
            case 'reasoning':
              stick();
              assistant().reasoning += ev.text;
              break;
            case 'tool_call':
              stick();
              cur = null;
              items.push({
                kind: 'tool',
                callId: ev.call_id,
                name: ev.name,
                origin: ev.origin,
                args: ev.args,
                status: ev.needs_confirm ? 'pending' : 'running',
                output: '',
              });
              break;
            case 'tool_result': {
              const t = items.find((i) => i.kind === 'tool' && i.callId === ev.call_id);
              if (t) {
                t.output = ev.output;
                t.status = ev.denied ? 'denied' : ev.is_error ? 'error' : 'ok';
              }
              break;
            }
            case 'notice':
              stick();
              cur = null;
              items.push({ kind: 'notice', text: ev.text });
              break;
            case 'done': {
              chatId = ev.chat_id;
              const p = ev.total_prompt_tokens ?? 0;
              const o = ev.total_output_tokens ?? 0;
              const n = ev.api_calls ?? 1;
              last = { prompt: p, output: o, calls: n, eco: ev.eco };
              session = {
                calls: session.calls + n,
                prompt: session.prompt + p,
                output: session.output + o,
                messages: session.messages + 1,
              };
              if (ev.prompt_tokens != null) ctx = { used: ev.prompt_tokens, max: ctx?.max ?? null };
              break;
            }
            case 'failed':
              error = `${ev.code} : ${ev.message}${ev.retryable ? ' (réessayable)' : ''}`;
              break;
            case 'cancelled':
              cur = null;
              items.push({ kind: 'notice', text: 'Annulé.' });
              break;
          }
        },
      );
    } catch (e) {
      error = String(e);
    } finally {
      busy = false;
      loadChats();
      loadUsage();
      loadCtx();
    }
  }

  function stop() {
    if (chatId) api.cancelChat(chatId).catch((e) => (error = String(e)));
  }

  function onKey(e) {
    if (e.key === 'Tab' && suggestions.length) {
      e.preventDefault();
      pickSkill(suggestions[0].name);
    } else if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  }
</script>

<div class="shell">
  <aside class="side">
    <div class="brand">FS CODE</div>
    <button class="primary wide" onclick={newChat} disabled={busy}>+ Nouveau chat</button>
    <div class="chats">
      {#each chats as c (c.id)}
        <div class="chatrow" role="presentation" onmouseleave={() => confirmDel === c.id && (confirmDel = null)}>
          {#if renaming === c.id}
            <input
              class="renamein"
              use:focus
              bind:value={renameText}
              onkeydown={(e) => renameKey(e, c)}
              onblur={() => (renaming = null)}
            />
          {:else}
            <button class="chat" class:active={c.id === chatId} onclick={() => openChat(c.id)} title={c.title}>{c.title}</button>
            <div class="acts" class:show={confirmDel === c.id}>
              <button class="mini" title="Renommer" onclick={() => startRename(c)}>✎</button>
              <button class="mini danger" title="Supprimer" onclick={() => del(c)}>
                {confirmDel === c.id ? 'Supprimer ?' : '✕'}
              </button>
            </div>
          {/if}
        </div>
      {/each}
    </div>
    <div class="sidefoot">
      <button class="wide" onclick={() => openSettings('compte')} title="Compte, Code, Économie, Skills, MCP">
        ⚙ Réglages{#if eco}<span class="ecotag"> · ÉCO</span>{/if}
      </button>
    </div>
  </aside>

  <section class="main">
    <div class="top">
      <select bind:value={agentId} onchange={onAgent} disabled={busy} title="Agent">
        {#each agents as a (a.id)}
          <option value={a.id}>{a.name ?? a.id}</option>
        {/each}
      </select>

      <div class="seg">
        <button class:on={mode === 'chat'} onclick={() => setMode('chat')} disabled={busy}>Chat</button>
        <button class:on={mode === 'code'} onclick={() => setMode('code')} disabled={busy}>Code</button>
      </div>

      {#if mode === 'code'}
        <button class="folder" onclick={chooseFolder} disabled={busy} title={workdir ?? 'Choisir un dossier'}>
          {workdir ? '📁 ' + baseName(workdir) : '📁 Choisir un dossier…'}
        </button>
        <label class="chk" title="Valable pour les modifications de fichiers, jamais pour les commandes">
          <input type="checkbox" checked={autoAccept} onchange={onAuto} />
          Accepter auto les modifs
        </label>
      {/if}

      <span class="spacer"></span>
    </div>

    <div class="messages" bind:this={box}>
      {#if items.length === 0}
        <div class="empty">
          {mode === 'code' ? 'Mode Code : l’IA lit et modifie les fichiers du dossier choisi.' : 'Pose ta question.'}
        </div>
      {/if}
      {#each items as it, i (i)}
        {#if it.kind === 'user'}
          <div class="msg user"><div class="who">Moi</div><div class="body">{it.text}</div></div>
        {:else if it.kind === 'assistant'}
          <div class="msg">
            <div class="who">IA</div>
            <div class="body"><Message text={it.text} reasoning={it.reasoning} /></div>
          </div>
        {:else if it.kind === 'tool'}
          <Tool t={it} onresolve={resolve} />
        {:else}
          <div class="notice">{it.text}</div>
        {/if}
      {/each}
    </div>

    {#if error}<div class="error">{error}</div>{/if}

    <div class="composer">
      <div class="inputwrap">
        {#if suggestions.length}
          <div class="sugg">
            {#each suggestions as s (s.name)}
              <button onclick={() => pickSkill(s.name)}><b>${s.name}</b><small>{s.description}</small></button>
            {/each}
          </div>
        {/if}
        <textarea
          bind:this={ta}
          bind:value={input}
          onkeydown={onKey}
          rows="2"
          placeholder={mode === 'code' ? 'Demande quelque chose sur le projet…  ($ pour un skill)' : 'Écris ton message…  ($ pour un skill)'}
        ></textarea>
      </div>
      {#if busy}
        <button onclick={stop}>■ Stop</button>
      {:else}
        <button class="primary" onclick={send} disabled={!input.trim()}>Envoyer</button>
      {/if}
    </div>
  </section>

  <footer class="status">
    {#if budget}
      <span>Budget {budget.uncertain ? '~' : ''}{Math.round(budget.pct)} %</span>
      <div class="gauge" class:warn={budget.pct >= 80}><i style="width:{budget.pct}%"></i></div>
    {:else}
      <span>Budget –</span>
    {/if}
    <span class="sep">│</span>
    <span>Contexte {ctxInfo ? fmt(ctxInfo.used) + (ctxInfo.max ? ' / ' + fmt(ctxInfo.max) : '') : '–'}</span>
    {#if chatId}<button class="mini" onclick={compact} disabled={busy}>Compacter</button>{/if}
    <span class="sep">│</span>
    <span>Session {fmt(session.prompt + session.output)} tok · {session.calls} appels</span>
    <span class="spacer"></span>
    {#if eco}<span class="eco">ÉCO</span>{/if}
    <span>{modeLabel}</span>
  </footer>
</div>

{#if showSettings && settings}
  <Settings
    {settings}
    {usage}
    {workdir}
    {autoAccept}
    {eco}
    {last}
    {session}
    ctx={ctxInfo}
    {chatId}
    {busy}
    tab={settingsTab}
    onclose={closeSettings}
    onsaved={onSaved}
    oneco={setEco}
    onauto={setAuto}
    onpick={chooseFolder}
    oncompact={compact}
    onreset={resetSession}
    oninsert={onInsertSkill}
    onextchange={loadMcp}
  />
{/if}
