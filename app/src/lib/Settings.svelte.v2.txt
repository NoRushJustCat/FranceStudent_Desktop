<script>
  import * as api from './api.js';

  // settings : { baseUrl, hasKey, autoAccept, ... } ; onclose() ; onsaved()
  let { settings, onclose, onsaved } = $props();

  let baseUrl = $state(settings?.baseUrl ?? '');
  let apiKey = $state('');
  let autoAccept = $state(!!settings?.autoAccept);
  let busy = $state(false);
  let error = $state('');

  const canSave = $derived(!busy && (settings?.hasKey || apiKey.trim() !== ''));

  async function save() {
    busy = true;
    error = '';
    try {
      await api.saveSettings({ baseUrl, apiKey, autoAccept });
      onsaved?.();
    } catch (e) {
      error = String(e);
      busy = false;
    }
  }
</script>

<svelte:window onkeydown={(e) => e.key === 'Escape' && onclose?.()} />

<div class="overlay">
  <form class="dialog" onsubmit={(e) => { e.preventDefault(); save(); }}>
    <h2>Réglages</h2>

    <label>
      URL de l'API
      <input bind:value={baseUrl} spellcheck="false" />
    </label>

    <label>
      Clé API
      <input
        type="password"
        bind:value={apiKey}
        autocomplete="off"
        placeholder={settings?.hasKey ? 'enregistrée — laisser vide pour la conserver' : 'colle ta clé ici'}
      />
    </label>

    <label class="chk">
      <input type="checkbox" bind:checked={autoAccept} />
      Accepter auto les modifs (jamais pour les commandes)
    </label>

    {#if error}<div class="error">{error}</div>{/if}

    <div class="row">
      <button type="button" onclick={() => onclose?.()}>Annuler</button>
      <button type="submit" class="primary" disabled={!canSave}>Enregistrer</button>
    </div>
  </form>
</div>
