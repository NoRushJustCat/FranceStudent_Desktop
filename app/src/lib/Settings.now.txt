<script>
  import { untrack } from 'svelte';
  import { saveSettings } from './api.js';
  import Extensions from './Extensions.svelte';

  let {
    settings,
    usage = null,
    budget = null,
    workdir = null,
    autoAccept = false,
    eco = false,
    last = null,
    session = { calls: 0, prompt: 0, output: 0, messages: 0 },
    ctx = null,
    chatId = null,
    busy = false,
    tab: initialTab = 'compte',
    onclose,
    onsaved,
    oneco,
    onauto,
    onpick,
    oncompact,
    onreset,
    oninsert,
    onextchange
  } = $props();

  const TABS = [
    { id: 'compte', label: 'Compte', icon: '👤' },
    { id: 'code', label: 'Code', icon: '💻' },
    { id: 'eco', label: 'Économie', icon: '⚡' },
    { id: 'skills', label: 'Skills', icon: '✨' },
    { id: 'mcp', label: 'MCP', icon: '🔌' },
    { id: 'diag', label: 'Diagnostic', icon: '🩺' }
  ];

  let tab = $state(untrack(() => initialTab));
  $effect(() => {
    tab = initialTab;
  });

  let baseUrl = $state(untrack(() => settings?.baseUrl ?? ''));
  let apiKey = $state('');
  let showKey = $state(false);
  let saving = $state(false);
  let saved = $state(false);
  let err = $state('');
  let copied = $state(false);

  const fmt = (n) => Number(n ?? 0).toLocaleString('fr-FR');
  const clamp = (v) => Math.max(0, Math.min(100, Number(v) || 0));

  const current = $derived(TABS.find((t) => t.id === tab) ?? TABS[0]);
  const hasKey = $derived(!!settings?.hasKey);
  const pct = $derived(budget ? clamp(budget.pct) : 0);
  const ctxPct = $derived(ctx?.max ? clamp((ctx.used / ctx.max) * 100) : null);
  const isExt = $derived(tab === 'skills' || tab === 'mcp');
  const diag = $derived(
    JSON.stringify(
      { settings: { ...settings, apiKey: undefined }, usage, workdir, ctx, last, session },
      null,
      2
    )
  );

  async function save() {
    err = '';
    saved = false;
    saving = true;
    try {
      await saveSettings({
        baseUrl: baseUrl.trim() || null,
        apiKey: apiKey.trim() || null
      });
      apiKey = '';
      saved = true;
      setTimeout(() => (saved = false), 2500);
      await onsaved?.();
    } catch (e) {
      err = String(e?.message ?? e);
    } finally {
      saving = false;
    }
  }

  function keyDown(e) {
    if (e.key === 'Enter' && !saving && (hasKey || apiKey.trim())) save();
  }

  async function copy() {
    try {
      await navigator.clipboard.writeText(diag);
      copied = true;
      setTimeout(() => (copied = false), 1500);
    } catch {
      /* presse-papiers indisponible */
    }
  }
</script>

<div
  class="ov"
  role="presentation"
  onmousedown={(e) => {
    if (e.target === e.currentTarget) onclose?.();
  }}
>
  <div class="dlg" role="dialog" aria-modal="true" aria-label="Réglages">
    <nav class="side">
      <div class="ttl">Réglages</div>
      {#each TABS as t (t.id)}
        <button class="nav" class:on={tab === t.id} onclick={() => (tab = t.id)}>
          <span class="ic">{t.icon}</span>{t.label}
        </button>
      {/each}
    </nav>

    <section class="main">
      <header class="hd">
        <h2>{current.label}</h2>
        <button class="x" onclick={() => onclose?.()} aria-label="Fermer">✕</button>
      </header>

      <div class="swpane" class:ext={isExt}>
        {#if tab === 'compte'}
          <div class="card">
            <div class="lbl">URL de l’API</div>
            <input
              bind:value={baseUrl}
              placeholder="https://…"
              spellcheck="false"
              autocomplete="off"
              onkeydown={keyDown}
            />
          </div>

          <div class="card">
            <div class="row">
              <div class="lbl">Clé API</div>
              <span class="spacer"></span>
              {#if hasKey}
                <span class="badge ro">enregistrée</span>
              {:else}
                <span class="badge">absente</span>
              {/if}
            </div>
            <div class="row">
              <input
                class="grow"
                type={showKey ? 'text' : 'password'}
                value={apiKey}
                oninput={(e) => (apiKey = e.currentTarget.value)}
                onkeydown={keyDown}
                placeholder={hasKey ? 'Laisser vide pour conserver la clé actuelle' : 'Colle ta clé ici'}
                spellcheck="false"
                autocomplete="off"
              />
              <button onclick={() => (showKey = !showKey)}>{showKey ? 'Masquer' : 'Afficher'}</button>
            </div>
            <div class="hint">La clé est stockée localement et n’est jamais affichée après enregistrement.</div>
          </div>

          <div class="row">
            <button class="primary" onclick={save} disabled={saving || (!hasKey && !apiKey.trim())}>
              {saving ? 'Enregistrement…' : 'Enregistrer'}
            </button>
            {#if saved}<span class="okt">Enregistré ✓</span>{/if}
          </div>
          {#if err}<div class="error">{err}</div>{/if}

          {#if budget}
            <div class="card">
              <div class="row">
                <div class="lbl">Budget utilisé</div>
                <span class="spacer"></span>
                <span class="mono">{budget.uncertain ? '~' : ''}{Math.round(budget.pct)} %</span>
              </div>
              <div class="bar" class:warn={pct >= 80}><i style="width:{pct}%"></i></div>
              {#if budget.uncertain}
                <div class="hint">Valeur estimée : le format de la réponse d’usage n’est pas garanti.</div>
              {/if}
            </div>
          {/if}
        {:else if tab === 'code'}
          <div class="card">
            <label class="chk">
              <input
                type="checkbox"
                checked={autoAccept}
                onchange={(e) => onauto?.(e.currentTarget.checked)}
              />
              Accepter automatiquement les modifications de fichiers
            </label>
            <div class="hint">Désactivé : chaque modification demande ta confirmation dans la conversation.</div>
          </div>

          <div class="card">
            <div class="lbl">Dossier de travail (chat courant)</div>
            <div class="path mono">{workdir || 'Aucun dossier choisi'}</div>
            <div class="row">
              <button onclick={() => onpick?.()}>Changer…</button>
            </div>
          </div>
        {:else if tab === 'eco'}
          <div class="card">
            <label class="chk">
              <input type="checkbox" checked={eco} onchange={(e) => oneco?.(e.currentTarget.checked)} />
              Mode économie
            </label>
            <div class="hint">
              Moins d’étapes par message, sorties d’outils plus courtes, contexte compacté automatiquement.
              Réduit le nombre de requêtes API.
            </div>
          </div>

          <div class="card">
            <div class="lbl">Dernier message</div>
            {#if last}
              <div class="kv">
                <span>Tokens d’entrée</span><span>{fmt(last.prompt)}</span>
                <span>Tokens de sortie</span><span>{fmt(last.output)}</span>
                <span>Appels API</span><span>{fmt(last.calls)}</span>
                <span>Mode éco</span><span>{last.eco ? 'oui' : 'non'}</span>
              </div>
            {:else}
              <div class="hint">Aucun message envoyé depuis la dernière remise à zéro.</div>
            {/if}
          </div>

          <div class="card">
            <div class="row">
              <div class="lbl">Cumul de la session</div>
              <span class="spacer"></span>
              <button class="mini" onclick={() => onreset?.()}>Remettre à zéro</button>
            </div>
            <div class="kv">
              <span>Messages</span><span>{fmt(session?.messages)}</span>
              <span>Appels API</span><span>{fmt(session?.calls)}</span>
              <span>Tokens d’entrée</span><span>{fmt(session?.prompt)}</span>
              <span>Tokens de sortie</span><span>{fmt(session?.output)}</span>
            </div>
          </div>

          <div class="card">
            <div class="lbl">Contexte du chat</div>
            {#if ctx}
              <div class="kv">
                <span>Utilisé</span><span>{fmt(ctx.used)}{ctx.max ? ' / ' + fmt(ctx.max) : ''}</span>
              </div>
              {#if ctxPct !== null}
                <div class="bar" class:warn={ctxPct >= 80}><i style="width:{ctxPct}%"></i></div>
              {/if}
            {:else}
              <div class="hint">Contexte inconnu : ouvre ou démarre un chat.</div>
            {/if}
            <div class="hint">Compacter résume l’historique du chat pour libérer du contexte.</div>
            <div class="row">
              <button
                onclick={() => oncompact?.()}
                disabled={!chatId || busy}
                title={!chatId ? 'Aucun chat ouvert' : busy ? 'Génération en cours' : ''}
              >
                Compacter le contexte
              </button>
            </div>
          </div>
        {:else if tab === 'skills'}
          <Extensions embedded section="skills" {workdir} {oninsert} {onclose} onchange={onextchange} />
        {:else if tab === 'mcp'}
          <Extensions embedded section="mcp" {workdir} {onclose} onchange={onextchange} />
        {:else if tab === 'diag'}
          <div class="row">
            <div class="hint grow">Informations de diagnostic (sans la clé API). Utile pour signaler un bug.</div>
            <button onclick={copy}>{copied ? 'Copié ✓' : 'Copier'}</button>
          </div>
          <pre class="raw">{diag}</pre>
        {/if}
      </div>
    </section>
  </div>
</div>
