<script>
  import { saveSettings } from './api.js';
  import Extensions from './Extensions.svelte';

  // Fenêtre unique : Compte, Code, Économie, Skills, MCP, Diagnostic.
  // Les cases (éco, accepter auto) s'enregistrent tout de suite via App.
  let {
    settings,
    usage = null,
    workdir = null,
    autoAccept = false,
    eco = false,
    last = null,
    session = { calls: 0, prompt: 0, output: 0, messages: 0 },
    ctx = null,
    chatId = null,
    busy = false,
    tab: startTab = 'compte',
    onclose,
    onsaved,
    oneco,
    onauto,
    onpick,
    oncompact,
    onreset,
    oninsert,
    onextchange,
  } = $props();

  let tab = $state('compte');
  let baseUrl = $state('');
  let apiKey = $state('');
  let err = $state('');
  let saving = $state(false);
  let ok = $state('');

  // valeurs de départ (une seule fois, à l'ouverture)
  const init = () => ({ tab: startTab, url: settings?.baseUrl ?? '' });
  {
    const i = init();
    tab = i.tab;
    baseUrl = i.url;
  }

  const tabs = [
    ['compte', 'Compte'],
    ['code', 'Code'],
    ['eco', 'Économie'],
    ['skills', 'Skills'],
    ['mcp', 'MCP'],
    ['diag', 'Diagnostic'],
  ];
  const isExt = $derived(tab === 'skills' || tab === 'mcp');

  const fmt = (n) => (n == null ? '–' : n >= 1000 ? (n / 1000).toFixed(n >= 10000 ? 0 : 1) + 'k' : String(n));

  async function save() {
    saving = true;
    err = '';
    ok = '';
    try {
      await saveSettings({
        baseUrl: baseUrl.trim() || null,
        apiKey: apiKey.trim() || null,
      });
      apiKey = '';
      ok = 'Enregistré.';
      await onsaved?.();
    } catch (e) {
      err = String(e);
    }
    saving = false;
  }

  function onkey(e) {
    if (e.key === 'Escape' && !isExt) onclose?.();
  }
  function onback(e) {
    if (e.target === e.currentTarget && !isExt) onclose?.();
  }

  const diag = $derived(
    JSON.stringify({ settings: { ...settings, apiKey: undefined }, workdir, usage }, null, 2),
  );
</script>

<svelte:window onkeydown={onkey} />

<div class="overlay" role="presentation" onmousedown={onback}>
  <div class="dialog sw" role="dialog" aria-modal="true" aria-label="Réglages">
    <div class="swhead">
      <h2>Réglages</h2>
      <button class="x" onclick={() => onclose?.()} aria-label="Fermer" title="Fermer">✕</button>
    </div>

    <div class="swbody">
      <nav class="swnav">
        {#each tabs as [id, label] (id)}
   <button class:on={tab === id} onclick={() => (tab = id)}>{label}</button>
        {/each}
      </nav>

      <div class="swpane" class:ext={isExt}>
        {#if tab === 'compte'}
   <label>
     URL de l'API
     <input bind:value={baseUrl} spellcheck="false" placeholder="https://api.francestudent.org" />
          </label>
   <label>
     Clé API {#if settings?.hasKey}<span class="badge ro">enregistrée</span>{/if}
     <input
       type="password"
       bind:value={apiKey}
       autocomplete="off"
       placeholder={settings?.hasKey ? 'Laisser vide pour la conserver' : 'Colle ta clé ici'}
            />
          </label>
          <div class="hint">La clé n'est jamais réaffichée.</div>
          {#if err}<div class="error">{err}</div>{/if}
          {#if ok}<div class="hint good">{ok}</div>{/if}
   <div class="btns">
     <button class="primary" onclick={save} disabled={saving}>Enregistrer</button>
   </div>
        {:else if tab === 'code'}
          <label class="chk">
            <input type="checkbox" checked={autoAccept} onchange={(e) => onauto?.(e.currentTarget.checked)} />
     Accepter automatiquement les modifications de fichiers
          </label>
   <div class="hint">
     Valable pour <code>edit_file</code> et <code>write_file</code> seulement. Les commandes
     (<code>run_command</code>) demandent toujours confirmation.
          </div>
   <h3>Dossier de travail</h3>
   <div class="mono path">{workdir ?? 'aucun'}</div>
   <div class="btns left">
     <button onclick={() => onpick?.()} disabled={busy}>Choisir un dossier…</button>
          </div>
        {:else if tab === 'eco'}
   <label class="chk">
     <input type="checkbox" checked={eco} onchange={(e) => oneco?.(e.currentTarget.checked)} />
     <b>Mode économie</b>
          </label>
          <ul class="hint">
     <li>Sorties d'outils plafonnées à 8 000 caractères.</li>
     <li>10 étapes d'outils maximum par message.</li>
     <li>Descriptions d'outils et de skills raccourcies.</li>
     <li>Compaction automatique du contexte quand il devient gros.</li>
   </ul>

   <h3>Dernier message</h3>
   {#if last}
     <div class="kv">
       <span>Entrée</span><span>{fmt(last.prompt)} tok</span>
       <span>Sortie</span><span>{fmt(last.output)} tok</span>
       <span>Appels API</span><span>{last.calls}</span>
     </div>
          {:else}
            <div class="hint">Rien d'envoyé pour l'instant.</div>
   {/if}

   <h3>Session</h3>
   <div class="kv">
     <span>Messages</span><span>{session.messages}</span>
     <span>Entrée</span><span>{fmt(session.prompt)} tok</span>
     <span>Sortie</span><span>{fmt(session.output)} tok</span>
     <span>Appels API</span><span>{session.calls}</span>
     <span>Contexte du chat</span>
            <span>{ctx ? fmt(ctx.used) + (ctx.max ? ' / ' + fmt(ctx.max) : '') : '–'}</span>
   </div>
          <div class="btns left">
     <button onclick={() => oncompact?.()} disabled={!chatId || busy}>Compacter le contexte</button>
     <button onclick={() => onreset?.()}>Remettre à zéro</button>
   </div>
        {:else if tab === 'skills' || tab === 'mcp'}
   <Extensions embedded section={tab} {workdir} {oninsert} onchange={onextchange} onclose={() => onclose?.()} />
        {:else}
   <div class="hint">Données brutes, utiles pour corriger l'interface. La clé API n'y figure pas.</div>
   <pre class="raw">{diag}</pre>
 {/if}
      </div>
    </div>
  </div>
</div>
