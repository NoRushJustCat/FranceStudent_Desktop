<script>
  // t : {callId, name, origin, args, status: pending|running|ok|error|denied, output}
  // onresolve(t, allow, always)
  let { t, onresolve } = $props();

  let always = $state(false);

  const a = $derived(t.args && typeof t.args === 'object' ? t.args : {});
  const isMcp = $derived(t.origin === 'mcp');
  const isSkill = $derived(t.origin === 'skill');

  // mcp__serveur__outil -> « serveur · outil »
  const label = $derived.by(() => {
    if (!t.name.startsWith('mcp__')) return t.name;
    const [, srv, ...rest] = t.name.split('__');
    return `${srv} · ${rest.join('__')}`;
  });
  const target = $derived(a.path ?? a.pattern ?? a.command ?? a.name ?? '');
  const labels = { pending: 'en attente', running: 'en cours', ok: 'ok', error: 'erreur', denied: 'refusé' };

  const lines = (s) => String(s ?? '').split('\n').slice(0, 40);
  const preview = (s) => {
    const x = String(s ?? '');
    return x.length > 1500 ? x.slice(0, 1500) + '\n…' : x;
  };
  const hasArgs = $derived(Object.keys(a).length > 0);
</script>

<div class="tool {t.status}">
  <div class="thead">
    <b>{label}</b>
    {#if isMcp}<span class="badge mcp">mcp</span>{/if}
    {#if isSkill}<span class="badge skill">skill</span>{/if}
    <span class="tgt" title={String(target)}>{target}</span>
    <span class="st">{labels[t.status] ?? t.status}</span>
  </div>

  {#if t.name === 'run_command'}
    <pre class="cmd">$ {a.command}</pre>
  {:else if t.name === 'edit_file'}
    <pre class="diff">{#each lines(a.old_string) as l}<span class="del">- {l}</span>{/each}{#each lines(a.new_string) as l}<span class="add">+ {l}</span>{/each}</pre>
  {:else if t.name === 'write_file'}
    <pre>{preview(a.content)}</pre>
  {:else if (isMcp || isSkill) && hasArgs}
    <pre>{preview(JSON.stringify(a, null, 2))}</pre>
  {/if}

  {#if t.status === 'pending'}
    <div class="ask">
      <button class="primary" onclick={() => onresolve(t, true, always)}>Autoriser</button>
      <button onclick={() => onresolve(t, false, false)}>Refuser</button>
      {#if isMcp}
        <label class="chk"><input type="checkbox" bind:checked={always} /> Toujours autoriser (session)</label>
      {/if}
    </div>
  {/if}

  {#if t.output}
    <details class="out">
      <summary>Résultat</summary>
      <pre>{t.output}</pre>
    </details>
  {/if}
</div>
